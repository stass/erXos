# The seconds a configured domain's init has to report ready to the domain
# service: its row's boot_timeout_s, which the manifest source may leave out
# for the default, and which the reader bounds to 1 .. boot_timeout_max_s, so
# its milliseconds fit an Erlang timer. The default is also a session's and a
# create request's. It suits an emulated machine as well as hardware: QEMU
# virt under TCG on a loaded host takes longer than 10 s to boot a BeamAsm
# domain, and a longer default only notices a hung boot later.
limit boot_timeout_default_s u32 60
limit boot_timeout_max_s u32 3600
