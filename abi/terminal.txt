# The rules of the line escape, which renders text attributed to a source
# (log records, names, inspector output) on one visible line. A program's own
# output is not such text and does not pass through it. Tab and newline are
# controls (Cc), so the line escape renders them like any other.
terminal 1
unicode 17.0.0
input unicode_data otp/lib/stdlib/uc_spec/UnicodeData.txt
input properties otp/lib/stdlib/uc_spec/DerivedCoreProperties.txt
unsafe_categories Cc Cf Zl Zp Cn
unsafe_property Default_Ignorable_Code_Point
exceptions 200C 200D FE0E FE0F
invalid_scalars D800 DFFF
kernel_ascii 0020 007E
token codepoint <U+ 4 6 >
token byte <0x 2 2 >
token kernel \x 2 2 -
