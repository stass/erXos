# Profiles are separate: renderer output never grants permission to raw text.
# Tab is layout only in strict, whose text already holds newlines; a line of
# attributed text and the kernel log escape it, since it moves the cursor.
terminal 1
unicode 17.0.0
input unicode_data otp/lib/stdlib/uc_spec/UnicodeData.txt
input properties otp/lib/stdlib/uc_spec/DerivedCoreProperties.txt
input prim_tty otp/lib/kernel/src/prim_tty.erl
input io_ansi otp/lib/stdlib/src/io_ansi.erl
renderer_operations cursor_up cursor_down cursor_backward cursor_forward erase_display tab clear format
renderer_styles cyan_background light_white bold
unsafe_categories Cc Cf Zl Zp Cn
unsafe_property Default_Ignorable_Code_Point
exceptions 200C 200D FE0E FE0F
invalid_scalars D800 DFFF
pass strict 0009 000A
pass line
kernel_ascii 0020 007E
token codepoint <U+ 4 6 >
token byte <0x 2 2 >
token kernel \x 2 2 -
grammar sequence_bytes=64 parameters=16 digits=5 separator=3B intermediates=20..2F

# for size-bound rules. Bounds and defaults may name current rows or columns.

parameter columns 0 1 columns 1

parameter erase_display 0 1 0..3 0
parameter erase_line 0 1 0..2 0
parameter scroll_region 0 2 rows,rows 1,rows

parameter mode_1049 1 1 1049 1049

sgr indexed selectors=38,48,58 mode=5 operands=1 maximum=255
sgr rgb selectors=38,48,58 mode=2 operands=3 maximum=255
