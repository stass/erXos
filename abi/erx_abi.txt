# A pipe (docs/design-pipes.md). A data record's payload is 1 to pipe_buf
# bytes, a control record's whole tickets of pipe_ticket bytes, at most
# pipe_control_max. A pipe has at most pipe_readers_max read ends, one bit
# each in the reader page's sleepers and the lock page's waiters. Write
# ends are numbered from 1, so pipe_none, the lock page's "no current
# record", is what a fresh page of zeros reads. A pipe's first write end's
# ring has a data area of pipe_ring_first bytes, every other's
# pipe_ring_next; the area starts pipe_ring_data bytes into the ring's
# memory object, past its header page.
limit pipe_buf u32 4096
limit pipe_ticket u32 8
limit pipe_control_max u32 128
limit pipe_readers_max u32 64
# A read end's largest index, pipe_readers_max - 1, which the program
# protocol's read ends name (protocols/prog.txt); liberx/erx_pipe_check.c
# fails to build unless the two agree.
limit pipe_rix_max u32 63
# A session's pipes quota: the most lanes its pipes hold at once
# (docs/design-pipes.md, Limits). The pipe server holds a session to it,
# and the program port sizes a program's credit for {drained} notices by
# it, since the gone writers of the read ends a program holds are lanes
# of its session's pipes (docs/design-programs.md, Credit).

limit pipe_none u32 0
limit pipe_ring_first u32 65536
limit pipe_ring_next u32 16384
limit pipe_ring_data u32 4096
# A pipe lane's reader page: the reader page of every lane, rpos and wake,
# stored by the read end that holds the reader lock, then the read ends
# that wait for data, bit r set and cleared by read end r. The write end
# maps it read-only.

  field wake u64 8
  field sleepers u64 16
  atomic rpos load acquire store release
  atomic wake load acquire store release
  atomic sleepers load acquire store release
  writer rpos consumer
  writer wake consumer
  writer sleepers readers
  owner rpos consumer all
  owner wake consumer all
  owner sleepers readers all
  visibility rpos release
  visibility wake release
  visibility sleepers release
# A pipe's lock page, which its read ends map, and the pipe server too,
# only to store wends and to free a gone read end's lock: the reader lock (0

  atomic cur_off load relaxed store relaxed
  atomic cur_pos load relaxed store relaxed

  writer wends pipe_server
  writer waiters readers
  writer cur holder
  writer cur_off holder
  writer cur_pos holder
  owner owner readers all
  owner wends pipe_server all
  owner waiters readers all
  owner cur holder all
  owner cur_off holder all
  owner cur_pos holder all
  visibility owner release
  visibility wends release
  visibility waiters release
  visibility cur release
  visibility cur_off release
  visibility cur_pos release
