# The Phase 1 root service's wire, in a native UTCB's message registers
# (docs/design-stopgap.md, "The root service"). A request's tag label is its
# fiasco_call number and its first word names the calling thread; a layout's
# size fixes the request's word count, which the service's decoder checks. A
# reply is erx_fiasco_reply. Number 1 is the ABI's debug_log.
enum fiasco_call u32
  value thread_create 2
  value thread_start 3
  value vm_reserve 4
  value vm_commit 5
  value vm_decommit 6
  value vm_release 7
  value space_protect 8
  value machine_end 9
  value thread_set_affinity 10
  value thread_reap 11
