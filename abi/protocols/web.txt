# The web protocol (docs/design-web.md, "The protocol"): every message a
# browser and the machine exchange, one WebSocket binary message each, and
# the bounds both ends hold them to. tools/lib/erx/abi_emit_web.rb generates
# erx_web_proto, the Erlang codec, and web/src/generated/proto.ts, the
# TypeScript codec, from it, and writes into both one SHA-256 over this
# file, ui.txt, inspection.txt, terminal.txt and the Unicode data
# terminal.txt names (docs/design-abi.md, "Web protocol").
#
# Members of "web VERSION", each a type: "integer NAME u8|u16|u32 [min N]";
# "bytes NAME MAX [exact]" and "text NAME MAX", a u32 length and that many
# bytes, which TypeScript reads as bytes and as MachineText; "digits NAME
# MAX", a decimal integer in its canonical form, which TypeScript reads as
# a bigint; "stopgap NAME", the name of a stopgap reporter the inspection
# schema declares; "ui_record NAME MAX", one record of ui.txt's binary form,
# its u32 length included, which ui.txt's own readers decode; "list NAME
# ELEMENT MAX", a u32 count and the elements; "tuple NAME FIELD:TYPE..." and
# "map NAME FIELD:TYPE...", the fields in order; "union NAME [depth N]
# [message MAX]", a u8 tag and the fields of its case, with one "case NAME
# TAG [FIELD:TYPE...]" beneath it per member. "boolean" is a u8 of 0 or 1.
# A MAX is a limit's name or a number. A union with "depth N" may name
# itself through its cases' fields; at level N its cases that do are
# unknown tags. A union with "message MAX" is a whole message of at most
# MAX bytes, with no bytes after its fields.
#
# Readings where docs/design-web.md states no bound: a text datum, the ui
# operations of an update and a record's key, kind and attributes are
# bounded by the message alone or by the datum's own bounds as named below;
# an attribute key and a kind are atom names; a record's children are a
# datum list; an entry's time is a decimal integer; a trace selection's
# presentation is its type's name and its canonical text.

limit web_message_max u32 65536
# One ui.txt record inside an update or an event: the message less 512 bytes
# kept for the update's header and the list's count. It bounds the record's
# length word, which counts the tag and the fields and not itself, as
# ui.txt's own readers do; ui.txt's insert bound is checked against it.
limit web_ui_op_max u32 65024
limit web_views_max u32 16
limit web_path_segments_max u32 16
limit web_segment_max u32 256
limit web_trace_selection_max u32 16
limit web_entries_max u32 1000
limit web_uis_max u32 256
limit web_name_max u32 255
limit web_build_id_max u32 64
limit web_int_max u32 40
limit web_atom_max u32 255
limit web_printed_max u32 4096
limit web_list_max u32 1000
limit web_tuple_max u32 16
limit web_datum_depth u32 8
limit web_ping_ms u32 30000
limit web_idle_ms u32 60000

web 1
  integer view u32
  integer ui_number u32
  integer count u32
  integer timeout_ms u32 min 1
  bytes hash 32 exact
  bytes segment web_segment_max
  bytes type_name web_atom_max
  bytes presented web_printed_max
  text name web_name_max
  text build_id web_build_id_max
  text atom_name web_atom_max
  text printed web_printed_max
  text reported web_message_max
  digits int web_int_max
  stopgap stopgap_name
  ui_record ui_op web_ui_op_max
  list segments segment web_path_segments_max
  tuple selected type:type_name text:presented
  list selection selected web_trace_selection_max
  union target
    case path 1 segments:segments
    case trace 2 selection:selection frames:boolean kernel:boolean
    case ui 3 number:ui_number
    case uis 4
  union browser message web_message_max
    case hello 1 hash:hash
    case open 2 view:view target:target
    case next 3 view:view
    case close 4 view:view
    case event 5 view:view record:ui_op
  union datum depth web_datum_depth
    case int 1 digits:int
    case text 2 bytes:reported
    case atom 3 name:atom_name
    case present 4 type:atom_name text:printed
    case list 5 items:datum_list
    case tuple 6 items:datum_tuple
    case printed 7 text:printed
  list datum_list datum web_list_max
  list datum_tuple datum web_tuple_max
  list path datum web_path_segments_max
  union reporter
    case kernel 1
    case service 2
    case runtime 3
    case stopgap 4 name:stopgap_name
  union status
    case ok 1
    case partial 2
    case unknown 3
    case unsupported 4
    case timeout 5 ms:timeout_ms
    case error 6
  tuple attr key:atom_name reporter:reporter status:status value:datum
  list attrs attr web_list_max
  union completeness
    case complete 1
    case partial 2
    case unknown 3
  tuple children completeness:completeness names:datum_list
  map wire_record path:path kind:atom_name id:datum attrs:attrs children:children
  union entry
    case entry 1 time:int value:datum
    case gap 2 count:count
  list entries entry web_entries_max
  list ui_ops ui_op web_message_max
  tuple open_ui number:ui_number label:name
  list uis open_ui web_uis_max
  union body
    case record 1 record:wire_record
    case entries 2 entries:entries
    case ui 3 ops:ui_ops
    case uis 4 uis:uis
  union bye_reason
    case mismatch 1
    case protocol 2
    case limit 3
    case logout 4
    case expired 5
    case shutdown 6
  union machine message web_message_max
    case welcome 1 principal:name build:build_id
    case update 2 view:view body:body
    case ended 3 view:view reason:name
    case bye 4 reason:bye_reason
