# The program protocol's frame profile: the frames of a program's channel
# (docs/design-programs.md#the-startup-message,
# docs/design-programs.md#the-program-protocol) and their bounds
# (docs/design-abi.md#program-protocol). tools/lib/erx/abi_emit_protocol.rb
# generates the Erlang grammar build/gen/otp/erts/preloaded/src/erx_prog_grammar.erl
# from it, and tools/lib/erx/abi_emit_profile.rb the table liberx's C check
# reads (build/gen/include/erx/prog_profile.h and
# build/gen/liberx/generated/prog_profile.c).
#
# Readings where the programs and pipes designs state no bound:
# * A pid and a pgid are 2 to 2^31 - 1 and a ppid 1 to 2^31 - 1, as the
#   startup message gives them.
# * A pipe's number, an end's number, a write end's number and a pipe's
#   count of write ends, pos_integer() in the designs, are u32 from 1, as the
#   pipe server's C layouts hold them (docs/design-pipes.md#the-pipe-servers-records);
#   a read end's index is 0 to pipe_rix_max, and a write end's list of
#   read ends holds at most pipe_readers_max of them and a read end's,
#   the others, at most pipe_rix_max.
# * A read end's list of write ends holds at most 8,192 entries, more than
#   a frame of prog_frame_max bytes can carry, so the frame bound is the
#   one that holds it.
# * The data area's Slots, its count of 64 KiB slots, is 1 to 16, the
#   slots a request may name.
# * An open is any u16, an entry of the program's open table.
# * A mem descriptor's range is in pages of erx_abi.txt's page_size, a first
#   page from 1 and a count from 1, each a u32, since every u64 of a frame
#   is a capability: the mapping starts below 2^44 bytes and ends below 2^45, and
#   page 0 is never mapped.
#
# Members (docs/design-abi.md#program-protocol has the full table):
# "version N" is the version every frame carries; "reader c" makes the
# generator emit the C check; "frame NAME to|from TAG [slot] KIND" is a
# frame {TAG, Version, Value} or {TAG, Version, Slot, Value}. An integer is
# u8, u16, u32 or u64, "in MIN MAX" or "values N...", "except N",
# "unique" (no two values of the kind in one frame are equal) and
# "refers LIST" (the value is the first field of an element of LIST that
# holds no refers value itself). A text has "max N", "utf8",
# "refuse nul", "assign" (a name before its first "="), "path" (absolute
# and normalised) and "sum TOTAL"; "total NAME MAX" bounds the texts marked
# "sum NAME", each counting its length and its NUL. A list has "max N" and
# "ascending"; a map's "field" keys must be present and its "optional"
# keys may be; a union's "case ATOM [=ATOM...] [KIND...]" is its atom, or
# a tuple of its atom, its literal atoms and its fields. "boolean" and
# "signal" are built in. "capability NAME" declares a capability handle:
# a u64 other than cap_null; no two capabilities of a frame are equal.
# A handle is 64 bits, so every other u64 kind or field fails, as does an
# integer excepting cap_null. For a union a frame to the program carries,
# the generator walks each case's capabilities and refuses one nested
# inside another kind.

limit prog_frame_max u32 196608
limit prog_args_max u32 131072
limit prog_path_max u32 4096
limit prog_fds_max u32 1024
limit prog_data_slots_max u32 16
# Each slot of a program's data area, in bytes: the session sizes the area
# and the program maps it as data_slots of these.
limit prog_data_slot_bytes u32 65536
# The labels of the doorbells bound into a program's group. The channel's
# bell takes prog_channel_label. A pipe end's takes
# (Pipe << pipe_label_shift) | End, computed in 64 bits, which is at least
# 2^32 because Pipe is at least 1, so no pipe end's label is the channel's.
limit prog_channel_label u64 1
limit pipe_label_shift u64 32
# The bytes of a frame the session writes after the startup message, a
# report: the executor copies each whole into a buffer of this size and
# refuses a longer one. The profile's largest report, {erx_prog, 1,
# {pipe_writer, ...}} with three capabilities as 64-bit integers, takes
# about 120 bytes of ETF.
limit prog_report_max u32 512
# A program's buffer place (docs/design-display.md, "The pixel area"): a
# window's buffer leased to a program is mapped from prog_mem_base, for at
# most prog_mem_bytes, the largest buffer, 16,384 pixels square at 4 bytes
# a pixel. The place's first page number is below 2^32, as the startup
# message needs, and the place lies inside the stopgap's task range, 2^38 to
# 2^39, above the address a program is linked at.
limit prog_mem_base u64 0x7f00000000
limit prog_mem_bytes u64 1073741824

protocol erx_prog
  version 1
  reader c
  integer process_id u32 in 2 0x7fffffff
  integer parent_id u32 in 1 0x7fffffff
  integer fd u16 in 0 1023
  integer dup_target u16 in 0 1023 refers fds
  capability cap
  integer open u16 in 0 0xffff
  integer umask u16 in 0 511
  integer data_slots u8 in 1 @prog_data_slots_max
  integer pipe_id u32 in 1 0xffffffff
  integer end_id u32 in 1 0xffffffff
  integer wend u32 in 1 0xffffffff
  integer wend_count u32 in 1 0xffffffff
  integer rix u8 in 0 @pipe_rix_max
  integer lane_size u32 values 16384 65536
  integer mem_page u32 in 1 0xffffffff
  integer mem_pages u32 in 1 0xffffffff
  total args @prog_args_max
  text arg max @prog_args_max utf8 refuse nul sum args
  text env_var max @prog_args_max utf8 refuse nul assign sum args
  text cwd max @prog_path_max utf8 refuse nul path
  list argv arg max @prog_args_max
  list env env_var max @prog_args_max
  list signals signal max 128 ascending
  union access
    case read
    case write
    case read_write
  union end_state
    case open
    case gone
  tuple pipe_end pipe_id end_id
  tuple reader rix cap
  list write_readers reader max @pipe_readers_max ascending
  list read_readers reader max @pipe_rix_max ascending
  union writer_state
    case open cap
    case gone
  tuple writer wend cap cap lane_size writer_state
  list writers writer max 8192 ascending
  map write_end
    field wend wend
    field ring cap
    field rpage cap
    field size lane_size
    field readers write_readers
  map read_end
    field rix rix
    field lock cap
    field wends wend_count
    field writers writers
    field readers read_readers
  union descriptor
    case pipe =write pipe_end write_end
    case pipe =read pipe_end read_end
    case socketpair pipe_end read_end pipe_end write_end
    case file open access
    case tty open access
    case dir open
    case socket open
    case mem mem_page mem_pages
    case dup dup_target
    case null
  tuple fd_entry fd descriptor
  list fds fd_entry max @prog_fds_max ascending
  tuple data_area cap data_slots
  union image
    case object cap
  map start
    field pid process_id
    field ppid parent_id
    field pgid process_id
    field argv argv
    field env env
    field cwd cwd
    field spawn boolean
    field sigmask signals
    field sigignore signals
    field umask umask
    field data_area data_area
    field fds fds
    optional image image
  union request
  union notice
    case closed pipe_id end_id
    case drained pipe_id end_id wend
  union report
    case pipe_writer pipe_id end_id wend cap cap lane_size cap end_state
    case pipe_writer_gone pipe_id end_id wend
    case pipe_reader pipe_id end_id rix cap end_state
    case pipe_reader_gone pipe_id end_id rix
    case pipe_lost pipe_id end_id
    case pipe_server_lost
  frame start to erx_start start
  frame request from erx_prog slot request
  frame notice from erx_prog notice
  frame report to erx_prog report
