# EROFS's on-disk format, the subset erx_erofs reads (docs/design-storage.md,
# "Reading EROFS images"), from the format's MIT-licensed header erofs_fs.h:
# the superblock at its fixed offset, compact and extended inodes, directory
# entries and the inline extended-attribute header the reader steps over.
limit erofs_magic u32 0xe0f5e1e2
limit erofs_super_offset u32 1024
limit erofs_block_bits u32 12
limit erofs_block_size u32 4096
# An inode number counts 32-byte slots from the metadata area's first block.
limit erofs_inode_slot_bits u32 5
limit erofs_name_max u32 255
# Compatible features: the reader checks the superblock's CRC-32C when the
# first is set and takes compact inodes' own mtime when the second is.
limit erofs_compat_sb_chksum u32 0x1
limit erofs_compat_mtime u32 0x2
# The inode format word: bit 0 the inode layout, bits 1-3 the data layout,
# bit 4 "nlink is 1"; the reader refuses any higher bit.
limit erofs_i_version_mask u16 0x1
limit erofs_i_datalayout_shift u16 1
limit erofs_i_datalayout_mask u16 0x7
limit erofs_i_nlink_1 u16 0x10
limit erofs_i_all u16 0x1f
limit erofs_inode_layout_compact u16 0
limit erofs_inode_layout_extended u16 1
limit erofs_inode_flat_plain u16 0
limit erofs_inode_flat_inline u16 2
# The file type bits of i_mode, POSIX's, which EROFS stores as Linux does.
limit erofs_s_ifmt u16 0xf000
limit erofs_s_ifdir u16 0x4000
limit erofs_s_ifreg u16 0x8000
limit erofs_s_iflnk u16 0xa000
# The inline extended-attribute area of an inode with i_xattr_icount > 0 is
# the header and then icount - 1 words of 4 bytes.
limit erofs_xattr_entry_size u32 4
# The most blocks one erx_verity:read/3 call returns.
limit verity_read_blocks_max u32 16
layout erx_erofs_super_block 1 wire_le 8 128
  field magic u32 0
  field checksum u32 4
  field feature_compat u32 8
  field blkszbits u8 12
  field sb_extslots u8 13
  field root_nid u16 14
  field inos u64 16
  field epoch u64 24
  field fixed_nsec u32 32
  field blocks u32 36
  field meta_blkaddr u32 40
  field xattr_blkaddr u32 44
  field uuid u8 48 count 16
  field volume_name u8 64 count 16
  field feature_incompat u32 80
  field available_compr_algs u16 84
  field extra_devices u16 86
  field devt_slotoff u16 88
  field dirblkbits u8 90
  field xattr_prefix_count u8 91
  field xattr_prefix_start u32 92
  field packed_nid u64 96
  field xattr_filter_reserved u8 104
  padding 105 3
  field build_time u32 108
  field root_nid_8b u64 112
  padding 120 8
layout erx_erofs_inode_compact 1 wire_le 4 32
  field i_format u16 0
  field i_xattr_icount u16 2
  field i_mode u16 4
  field i_nlink u16 6
  field i_size u32 8
  field i_mtime u32 12
  field i_u u32 16
  field i_ino u32 20
  field i_uid u16 24
  field i_gid u16 26
  padding 28 4
layout erx_erofs_inode_extended 1 wire_le 8 64
  field i_format u16 0
  field i_xattr_icount u16 2
  field i_mode u16 4
  field i_nb u16 6
  field i_size u64 8
  field i_u u32 16
  field i_ino u32 20
  field i_uid u32 24
  field i_gid u32 28
  field i_mtime u64 32
  field i_mtime_nsec u32 40
  field i_nlink u32 44
  padding 48 16
# Entries sit 12 bytes apart, so the 64-bit nid is declared as its two words.
layout erx_erofs_dirent 1 wire_le 4 12
  field nid_lo u32 0
  field nid_hi u32 4
  field nameoff u16 8
  field file_type u8 10
  padding 11 1
layout erx_erofs_xattr_ibody_header 1 wire_le 4 12
  field h_name_filter u32 0
  field h_shared_count u8 4
  padding 5 7
