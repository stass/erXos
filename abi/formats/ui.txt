# The display's widget trees (docs/design-abi.md, "Widget format", and
# docs/design-display.md, "Widget trees"): value types, image types,
# properties, kinds and the records of the binary form. Value types and
# image types share one namespace, properties one and kinds one; image types, properties and kinds each
# number from 1 to 255. A kind takes the properties marked common and those
# listed beneath it. tools/gen_abi.rb generates erx_ui_schema, the validator
# of whole trees, from it.

# The most children one node of a tree has: it bounds the difference of two
# trees and keeps every insert's and move's index within a u16.
limit ui_children u32 1024
# The most rectangles one damage report carries.
limit ui_damage_rects u32 16

# Placement of a node within the room it gets.
enum ui_align u8
  value start 0
  value center 1
  value end 2
  value fill 3
