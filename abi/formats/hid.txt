# HID's constants, the one home of every HID value erXos reads (docs/design-usb.md,
# "HID"): the boot interface's codes, the class descriptors and requests the HID
# function uses, the report descriptor's items and bounds, the usages it reads, and the
# keyboard page's map to evdev codes. A value erXos has no use for is not entered.
#
# The sources are the USB Device Class Definition for HID 1.11 and the HID Usage Tables
# 1.5 (sections cited), and FreeBSD's sources at release/15.1.0-p6 (95719d8a), under the
# BSD-2-Clause license: sys/dev/usb/usb.h, sys/dev/usb/usbhid.h, sys/dev/hid/hid.h,
# sys/dev/hid/hid.c, sys/dev/hid/hkbd.c and sys/dev/evdev/evdev_utils.c. A value FreeBSD
# defines is FreeBSD's; the others are the specification's or erXos's own, as their
# comments say.

# A boot interface (subclass 1) and its two protocols, keyboard and mouse, read from
# bInterfaceSubClass and bInterfaceProtocol (HID 1.11, sections 4.2 and 4.3; usb.h,
# lines 455-457).
limit hid_subclass_boot u8 1
enum hid_boot_protocol u8
  value keyboard 1
  value mouse 2

# The class descriptor types: the HID descriptor in the configuration, and the report
# descriptor it names (HID 1.11, section 7.1; usbhid.h, lines 40-41).
limit hid_descriptor_hid u8 0x21
limit hid_descriptor_report u8 0x22

# The class requests the function sends, as bRequest (HID 1.11, section 7.2;
# usbhid.h, lines 47 and 49), and SET_PROTOCOL's wValue: 0 for the boot protocol, as
# hkbd.c sends it (line 1015), and 1 for the report protocol (HID 1.11, section 7.2.6).
limit hid_request_set_idle u8 0x0a
limit hid_request_set_protocol u8 0x0b
enum hid_protocol u16
  value boot 0
  value report 1

# A report descriptor's items (HID 1.11, section 6.2.2; hid.c, lines 305-318): a short
# item's prefix holds its tag in bits 7-4, its type in bits 3-2 and its size in bits 1-0,
# 0, 1, 2 or 4 bytes of data, the last coded 3; the prefix 0xfe opens a long item, which
# the parser skips.
limit hid_item_long u8 0xfe
enum hid_item_type u8
  value main 0
  value global 1
  value local 2

# The tags of the items the parser reads, by type (hid.c, lines 353-408, 412-479 and
# 483-558). A tag of a known type that is not listed sets nothing the layout keeps.
enum hid_main_tag u8
  value input 8
  value output 9
  value collection 10
  value feature 11
  value end_collection 12
enum hid_global_tag u8
  value usage_page 0
  value logical_minimum 1
  value logical_maximum 2
  value report_size 7
  value report_id 8
  value report_count 9
  value push 10
  value pop 11
enum hid_local_tag u8
  value usage 0
  value usage_minimum 1
  value usage_maximum 2

# The data bits of an Input item that the layout keeps: a constant field, which is
# padding; a variable field, one value per usage, where unset is an array of usage
# indices; a relative field, where unset is absolute (HID 1.11, section 6.2.2.4; hid.h,
# lines 183-185).
flags hid_field_flags u32
  bit constant 0
  bit variable 1
  bit relative 2

# The bounds of a report descriptor. FreeBSD's: at most 64 usages a main item, 16
# report ids with id 0 among them, so 15 ids where a descriptor uses them, a report
# count of 2048, and a push stack of four levels, the current one included, so at most
# three pushes outstanding (hid.c, lines 66-69, 130-139 and 448). erXos's: a field of
# 1 to 32 bits, an input report of at most the 1 KiB interrupt buffer with its report id
# byte, and a descriptor of at most one 4 KiB control buffer (docs/design-usb.md, "What
# a device can make the domain spend").
limit hid_usages_max u32 64
limit hid_report_ids_max u32 16
limit hid_report_count_max u32 2048
limit hid_push_levels_max u32 4
limit hid_field_bits_max u32 32
limit hid_input_report_max u32 1024
limit hid_report_descriptor_max u32 4096

# The usage pages the function reads (HID Usage Tables 1.5, chapters 4, 10 and 12;
# hid.h, lines 35, 40 and 42).
enum hid_usage_page u16
  value generic_desktop 0x01
  value keyboard 0x07
  value button 0x09

# The Generic Desktop usages of a pointer and a wheel (hid.h, lines 77, 78 and 85).
enum hid_desktop_usage u16
  value x 0x30
  value y 0x31
  value wheel 0x38

# The Keyboard page's ErrorRollOver, which every array field of a report holds when
# more keys are down than it can carry (HID 1.11, Appendix C; hkbd.c, line 213), and
# its modifiers, the eight one-bit variables from Left Control to Right GUI (HID 1.11,
# Appendix B.1).
limit hid_key_error_roll_over u8 0x01
limit hid_key_modifier_first u8 0xe0
limit hid_key_modifier_last u8 0xe7

# The Button page's first three buttons, as FreeBSD's hms maps them from BTN_MOUSE,
# which is BTN_LEFT (docs/design-usb.md, "The report descriptor").
translation hid_button_key u8 evdev
  map 1 btn_left
  map 2 btn_right
  map 3 btn_middle
