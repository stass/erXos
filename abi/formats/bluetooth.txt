# Bluetooth's constants, the one home of every value the Bluetooth domain reads from or
# writes to its controller (docs/design-bluetooth.md): H4's packet indicators, the HCI
# commands and events the controller process uses, and the bounds a radio peer or the
# controller cannot pass. A value erXos has no use for is not entered.
#
# The source is the Bluetooth Core Specification 6.3, its HTML edition, Vol 4 Part A
# (UART transport) and Vol 4 Part E (HCI), sections cited; the hashes of the files read
# are in docs/design-bluetooth.md, "Sources". Bumble v0.0.237 (34def7f1, Apache-2.0),
# bumble/hci.py, lines 251-808, states the same codes. A value marked erXos's is the
# design's own choice, not the specification's.

# The HCI packet indicator before each packet on the line (Part A, section 2,
# Table 2.1). Synchronous and ISO data are never enabled, so the host meets their
# indicators only as a loss of synchronization (Part A, section 4).
enum h4_indicator u8
  value command 0x01
  value acl 0x02
  value synchronous 0x03
  value event 0x04
  value iso 0x05

# An opcode is the OGF in its upper 6 bits and the OCF in its lower 10 (Part E, section
# 5.4.1); erx_hci composes the two. The groups (7.1, 7.3, 7.4 and 7.8):
limit hci_ocf_bits u8 10
enum hci_ogf u8
  value link_control 0x01
  value controller 0x03
  value informational 0x04
  value le 0x08

# The OCF of each command used, by group: Disconnect (7.1.6); Set Event Mask, Reset and
# Host Buffer Size (7.3.1, 7.3.2, 7.3.39); Read Local Version Information, Read Local
# Supported Commands [v1], Read Buffer Size and Read BD_ADDR (7.4.1, 7.4.2, 7.4.5,
# 7.4.6); the LE commands [v1] (7.8.1 to 7.8.9); and LE Long Term Key Request Reply and
# Negative Reply (7.8.25, 7.8.26).
enum hci_link_control_ocf u16
  value disconnect 0x0006
enum hci_controller_ocf u16
  value set_event_mask 0x0001
  value reset 0x0003
  value host_buffer_size 0x0033
enum hci_informational_ocf u16
  value read_local_version 0x0001
  value read_local_commands 0x0002
  value read_buffer_size 0x0005
  value read_bd_addr 0x0009
enum hci_le_ocf u16
  value le_set_event_mask 0x0001
  value le_read_buffer_size 0x0002
  value le_read_features 0x0003
  value le_set_random_address 0x0005
  value le_set_advertising_parameters 0x0006
  value le_set_advertising_data 0x0008
  value le_set_scan_response_data 0x0009
  value le_set_advertising_enable 0x000a
  value le_long_term_key_request_reply 0x001a
  value le_long_term_key_request_negative_reply 0x001b

# The events the controller process decodes (7.7.5, 7.7.8 [v1], 7.7.14, 7.7.15, 7.7.16,
# 7.7.19, 7.7.39 and 7.7.65), and the LE Meta subevents among them (7.7.65.1, 7.7.65.3
# and 7.7.65.5). Any other event is counted and dropped.
enum hci_event u8
  value disconnection_complete 0x05
  value encryption_change 0x08
  value command_complete 0x0e
  value command_status 0x0f
  value hardware_error 0x10
  value number_of_completed_packets 0x13
  value encryption_key_refresh_complete 0x30
  value le_meta 0x3e
enum hci_le_subevent u8
  value connection_complete 0x01
  value connection_update_complete 0x03
  value long_term_key_request 0x05

# The bits of Set Event Mask's Event_Mask (7.3.1) and of LE Set Event Mask's
# LE_Event_Mask [v1] (7.8.1) that enable the events above; the bring-up sets these and
# no others. Command Complete, Command Status and Number Of Completed Packets cannot be
# masked.
enum hci_event_mask_bit u8
  value disconnection_complete 4
  value encryption_change 7
  value hardware_error 15
  value encryption_key_refresh_complete 47
  value le_meta 61
enum hci_le_event_mask_bit u8
  value connection_complete 0
  value connection_update_complete 2
  value long_term_key_request 4

# A status of success (Vol 1 Part F, as Part E, section 7, cites it for every command).
limit hci_status_success u8 0x00
# Unknown Connection Identifier (Vol 1 Part F, 2.2): a disconnect naming a handle the
# controller has already let go.
limit hci_status_unknown_connection u8 0x02
# Command Disallowed (Vol 1 Part F, 2.12): the controller cannot take the command now,
# which NimBLE's and Zephyr's controllers answer to a disconnect for a connection already
# ending (docs/design-bluetooth.md, "Commands and data credits").
limit hci_status_command_disallowed u8 0x0C
# Encryption Change's Encryption_Enabled for encryption on with AES-CCM on LE (7.7.8),
# the one value with which a link counts as encrypted.
limit hci_encryption_on_le u8 0x01
# The reasons the host gives Disconnect, among those 7.1.6 allows (Vol 1 Part F, 2.19,
# 2.20): Remote User Terminated Connection for a link that ended, and Remote Device
# Terminated Connection due to Low Resources for a link whose backlog reached
# bt_link_backlog (docs/design-bluetooth.md, "Commands and data credits").
enum hci_disconnect_reason u8
  value remote_user_terminated 0x13
  value low_resources 0x14
# LE Connection Complete's Peer_Address_Type (7.7.65.1): a public or a random device
# address; every other value is reserved.
enum hci_peer_address_type u8
  value public 0x00
  value random 0x01
# A Connection_Handle is the low 12 bits of its 16-bit word, below the two bits each of
# Packet_Boundary_Flag and Broadcast_Flag, and runs from 0x000 to 0xEFF (5.4.2).
limit hci_handle_bits u8 12
limit hci_handle_max u16 0x0eff
# LE Read Buffer Size reserves an LE_ACL_Data_Packet_Length from 0x0001 to 0x001A
# (7.8.2): every controller takes at least 27 octets of data (5.4.2).
limit hci_le_acl_length_min u16 27
# LE Set Advertising Data and LE Set Scan Response Data carry 31 octets (7.8.7, 7.8.8).
limit hci_advertising_data_max u8 31
# LE Set Advertising Parameters (7.8.5): connectable undirected advertising (ADV_IND),
# the random device address as the advertiser's own, all three channels (bits 0 to 2 of
# the channel map), and the interval's unit, 0.625 ms.
limit hci_advertising_type_ind u8 0x00
limit hci_own_address_random u8 0x01
limit hci_advertising_channels_all u8 0x07
limit hci_advertising_interval_unit_us u32 625
# The advertising and scan response data are AD structures, each a length, a type and its
# data (Vol 3 Part C, 11), of the types Bumble names (bumble/core.py, lines 1594 to 1622;
# unverified against the Core Specification Supplement): the Flags, complete lists of
# 16-bit and 128-bit service UUIDs, the shortened and the complete local name, and the
# appearance. The Flags erXos sends say LE General Discoverable and BR/EDR not supported.
enum bt_ad_type u8
  value flags 0x01
  value uuid16_complete 0x03
  value uuid128_complete 0x07
  value name_shortened 0x08
  value name_complete 0x09
  value appearance 0x19
limit bt_ad_flags_general_no_bredr u8 0x06

# erXos's bounds (docs/design-bluetooth.md, "What a radio peer can make the domain
# spend"): the longest ACL packet the host accepts, which Host Buffer Size states, 4
# octets of L2CAP header and ATT's MTU of 247; the most connections a row may allow; the
# most ACL packets in flight to the controller; the host's packet count that Host Buffer
# Size states, flow control from the controller staying off; and the advertising interval.
limit bt_acl_rx_max u16 251
limit bt_connections_max u8 4
limit bt_acl_credits_max u8 8
limit bt_host_acl_packets u16 8
limit bt_advertising_interval_us u32 100000
# The notifications a link holds waiting for its share of the credits (docs/design-bluetooth.md,
# "What a radio peer can make the domain spend").
limit bt_notify_queue u8 16
# The controller process's messages a link holds and has not acknowledged, events and ACL
# packets alike (docs/design-bluetooth.md, "Commands and data credits"); at the bound the
# link is sent nothing more and disconnected.
limit bt_link_backlog u8 32
# The store (docs/design-bluetooth.md, "Bonds and the store"): the most bonds it keeps, the
# largest stored state it writes or reads, the vault's state slot's 4 KiB, and the
# deadline of each call of its backend, after which a get is unavailable and a put
# unconfirmed.
limit bt_bonds_max u8 8
limit bt_state_max u16 4096
limit bt_store_timeout_ms u32 5000
# The pairing window (docs/design-bluetooth.md, "The pairing window"): the time a window
# admits a pairing; the longest first write of its `pair` file, `display` or `nodisplay`;
# and SMP's pairing wait (Vol 3 Part H, 2.3.6): the minimum for the first failure, then
# twice the last failure's wait halved for each whole decay interval since it, kept
# between the minimum and the maximum. A reader of the control tree's lines, a window's
# or the bonds list's (docs/design-bluetooth.md, "The client", erx_bt_pair:next/1), holds
# at most bt_pair_line_max + 1 octets that end no line; the longest line the window
# composes is 80.
limit bt_pair_window_s u32 120
limit bt_pair_write_max u32 16
limit bt_pair_line_max u32 128
limit bt_pair_wait_min_s u32 2
limit bt_pair_wait_max_s u32 64
limit bt_pair_wait_decay_s u32 600
# A failed pairing's reasons (docs/design-bluetooth.md, "Refusals and error results"),
# the words a `failed` line gives (erx_bt_pair): the Security Manager's, each mapped from
# one Pairing Failed code or one event, peer for the phone's own Pairing Failed, the
# store's, and the link's end. The Security Manager, the store and the window make each
# reason through a function guarded by ?ERX_IS_BT_PAIR_REASON, erx_bt_pair composes and
# reads a line's word against this set alone, and only the names cross: the values,
# which the enum syntax asks for, are never encoded.
enum bt_pair_reason u8
  value secure_connections_required 1
  value key_size 2
  value method 3
  value confirm 4
  value dhkey 5
  value bonding 6
  value invalid 7
  value unexpected 8
  value peer 9
  value timeout 10
  value already_bonded 11
  value bonds_full 12
  value unavailable 13
  value disconnected 14

# erXos's deadlines (docs/design-bluetooth.md, "The bring-up"): every command's answer,
# and the patch's two waits, NetBSD's (usr.sbin/btattach/init_bcm43xx.c, lines 123-125
# and 139), and the vendor command that starts the download, as the design cites it
# (docs/design-bluetooth.md, "The hardware").
limit bt_command_timeout_ms u32 5000
# The line's reset (docs/design-bluetooth.md, "A reset"): two power notifications, each
# answered within the mailbox's 1 s, and the 300 ms and 500 ms off and on times.
limit bt_line_reset_timeout_ms u32 5000
limit bt_patch_minidriver_ms u32 1000
limit bt_patch_launch_ms u32 4000
limit bt_patch_download_minidriver u16 0xfc2e
# A disconnect handed to the controller and not ended by a successful Disconnection
# Complete within this deadline ends the controller process (docs/design-bluetooth.md,
# "Commands and data credits"): it must stay above bt_command_timeout_ms plus the longest LE
# supervision timeout, 32 s, plus one longest connection interval, 4 s (Vol 4 Part E,
# 7.7.65.1), so a termination already under way has ended by then.
limit bt_disconnect_timeout_ms u32 45000

# The host's side of the LE-U link (docs/design-bluetooth.md, "Links: the fixed channels").
# An ACL packet's Packet_Boundary_Flag (Vol 4 Part E, 5.4.2): the host starts a frame it
# sends with first_non_flushable and the controller one it delivers with first_flushable,
# each continued with continuing; on LE-U neither side uses the other's start, and 0b11 is
# no longer used.
enum hci_acl_boundary u8
  value first_non_flushable 0
  value continuing 1
  value first_flushable 2
# The basic L2CAP header, PDU Length and Channel ID of 2 octets each (Vol 3 Part A, 3.1),

limit l2cap_header u8 4
enum l2cap_cid u16
  value att 0x0004
  value le_signaling 0x0005
  value smp 0x0006
# LE signaling (Part A, 4): a command's code, identifier and Data Length; the smallest

limit l2cap_le_mtu_sig u8 23

  value disconnection_rsp 0x07
  value connection_parameter_update_rsp 0x13
  value le_credit_based_connection_rsp 0x15
  value credit_based_connection_rsp 0x18
  value credit_based_reconfigure_rsp 0x1a
enum l2cap_reject_reason u16
  value not_understood 0x0000
  value mtu_exceeded 0x0001
# SMP's channel MTU with LE Secure Connections (Vol 3 Part H, 3.2, Table 3.2).
limit smp_mtu u8 65
