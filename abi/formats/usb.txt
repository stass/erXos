# The USB constants the stack reads (docs/design-usb.md): the bounds a device meets,
# chapter 9's requests, descriptors and features, the hub class's, and the Bulk-Only
# transport's with the SCSI commands it carries. HID's class constants are in hid.txt,
# the controller's memory structures in xhci.txt.
#
# Each value is the one FreeBSD gives, at releng/15.1 (commit 95719d8a895b), in the file
# and line cited: sys/dev/usb/usb.h, usb_freebsd.h, usb_device.c, usb_transfer.c,
# storage/umass.c and sys/cam/scsi/scsi_all.h. A value that is a choice of erXos's says
# so. Every multi-byte field on the bus is little-endian; every one in a SCSI command or
# its data is big-endian.

# What a device can make the domain spend (design-usb.md): the bounds a device meets,
# fixed before any device arrives; a device that offers more is refused. Each is erXos's
# choice unless a source is cited.
# The slots of one controller, those still being torn down included.
limit usb_devices_max u32 16
# Tiers of hubs below the root, FreeBSD's USB_HUB_MAX_DEPTH (usb_freebsd.h:90).
limit usb_hub_depth_max u32 5
# A hub's ports: the most one tier of the route string names (xhci.txt).
limit usb_hub_ports_max u32 15
# A configuration descriptor's wTotalLength: one control buffer.
limit usb_config_max u32 4096
# The endpoints a device may use, EP0 included.
limit usb_endpoints_max u32 8
# The input numbers the USB domain offers the display.
limit usb_input_max u32 8
# The drives the USB domain offers the file servers.
limit usb_drives_max u32 4
# The enumeration attempts a port starts in any usb_port_attempt_window_ms; past them
# a settled port waits until its oldest attempt leaves the window.
limit usb_port_attempts_max u32 3
limit usb_port_attempt_window_ms u32 60000
# The restarts of one function in any usb_function_restart_window_ms, counted as a
# port's attempts are; an end past them detaches the function's device
# (design-usb.md, "Failure handling").
limit usb_function_restarts_max u32 3
limit usb_function_restart_window_ms u32 60000
# The restarts a controller's bus group, erx_usb_bus_sup, makes in
# any usb_bus_restart_window_ms; past them the bus group ends and is not started again,
# so a controller that fails at every reset stops being reset (design-usb.md, "The
# process tree").
limit usb_bus_restarts_max u32 3
limit usb_bus_restart_window_ms u32 60000

# The lengths the stack asks GET_DESCRIPTOR for: a device descriptor's first 8 bytes,
# the packet size a full-speed EP0 starts at (USB_MAX_IPACKET, usb.h:70), then its
# whole 18 (struct usb_device_descriptor, usb.h:289-309); a configuration's 9-byte
# header (struct usb_config_descriptor, usb.h:386-400) before its wTotalLength; and a
# hub descriptor's 7 bytes before its port bitmaps (struct usb_hub_descriptor,
# usb.h:603-630).
limit usb_device_descriptor_head u32 8
limit usb_device_descriptor_size u32 18
limit usb_configuration_descriptor_size u32 9
limit usb_hub_descriptor_head u32 7
# A full-speed EP0's largest packet: its bMaxPacketSize0 is a power of two from
# usb_device_descriptor_head to this, the sizes FreeBSD's fallback tries
# (usbd_setup_device_desc, usb_request.c:1984-2026). At low, high and SuperSpeed EP0 has
# one packet size, so FreeBSD reads the whole device descriptor at once there.
limit usb_ep0_packet_full_max u16 64
# EP0's packet size when its device is addressed: the only size allowed at low, high and
# SuperSpeed, 8, 64 and 512, and at full speed the size its device descriptor's first 8
# bytes are read with until bMaxPacketSize0 is known (xHCI 1.2, 4.3, p. 85; FreeBSD's
# USB_MAX_IPACKET, usb.h:70, release/15.1.0, 95719d8a).
limit usb_ep0_packet_low u16 8
limit usb_ep0_packet_full_first u16 8
limit usb_ep0_packet_high u16 64
limit usb_ep0_packet_super u16 512
# The wait after a device's address is set before its next request, FreeBSD's
# usb_set_address_settle (usb.h:120), which usbd_req_set_address waits even when the
# controller sets the address itself (usb_request.c:1596-1599).
limit usb_set_address_settle_ms u32 10
# FreeBSD's port timing: a connected port's power-up delay before its reset, the
# recovery after the reset, and the time added to a hub's power-good time (usb.h:117-124);
# the unit of a hub descriptor's bPwrOn2PwrGood (usb.h:624), and the value FreeBSD's xHCI
# root hub descriptor gives it (controller/xhci.c:3533); the deadline of a port's reset,
# and the interval at which a resetting hub port's status is read again
# (usbd_req_reset_port, usb_request.c:805-863).
limit usb_power_up_delay_ms u32 300
limit usb_reset_recovery_ms u32 20
limit usb_extra_power_up_ms u32 20
limit usb_power_good_unit_ms u32 2
limit usb_root_power_good_units u32 10
limit usb_reset_deadline_ms u32 1000
limit usb_reset_poll_ms u32 50

limit usb_packet_extra_max u16 2
# FreeBSD's filter of standard packet sizes (usbd_get_std_packet_size,

# A SuperSpeed endpoint companion's bMaxBurst, zero-based: FreeBSD clamps the packet

# at most 3 KiB (xHCI 1.2, 4.14.2, p. 259, as design-usb.md cites it).

limit usb_ss_interrupt_payload_max u32 3072

# to bit 15 (USB 2.0, 11.24.2.3, Table 11-18). FreeBSD's usbd_req_clear_tt_buffer puts

# GET_STATUS's two words for a hub (usb.h:698-699) and for a port (usb.h:706-741),
# USB 2.0's bits. A hub's change word has its status word's bits. A root port's PORTSC

  bit over_current 1

  bit enabled 1

  bit over_current 3

  bit power 8
  bit low_speed 9
  bit high_speed 10
  bit test 11
  bit indicator 12

  bit connection 0
  bit enabled 1
  bit suspend 2
  bit over_current 3
  bit reset 4

# The Bulk-Only transport (umass.c:236-270): its two class requests, the 31-byte
# command block wrapper and the 13-byte command status wrapper with their signatures,
# the wrapper's direction flag, the longest command block it carries and the status
# a wrapper reports. Get Max LUN answers the highest unit's number; erXos refuses an
# interface answering more than 15 (design-usb.md, "The BOT engine"). Unverified: that
# 15 is the Bulk-Only specification's bound, which no source read here states; FreeBSD
# takes the answer whole (umass.c:1692-1700).
enum usb_bot_request u8
  value get_max_lun 0xfe
  value reset 0xff
limit usb_bot_cbw_size u32 31
limit usb_bot_cbw_signature u32 0x43425355
limit usb_bot_cbw_flag_in u8 0x80
limit usb_bot_cb_max u32 16
limit usb_bot_csw_size u32 13
limit usb_bot_csw_signature u32 0x53425355
limit usb_bot_lun_max u8 15
# The two wrong signatures a CSW of a device with the quirk wrong_csw_signature may carry
# in place of usb_bot_csw_signature, FreeBSD's CSWSIGNATURE_IMAGINATION_DBX1 and
# CSWSIGNATURE_OLYMPUS_C1 (umass.c:259-261, accepted at :1569-1576).
limit usb_bot_csw_signature_dbx1 u32 0x43425355
limit usb_bot_csw_signature_olympus u32 0x55425355
# The quirks of a mass-storage device that erXos's stack has a behaviour for, the closed
# set the rows of abi/formats/usb_quirks.txt may name (design-usb.md, "Device quirks").
# Each names its consumer and the behaviour it changes:
# force_bot, force_scsi: erx_usb_claim claims a mass-storage interface as Bulk-Only, or
#   as speaking SCSI, whatever its protocol, or its subclass, says; with both, an
#   interface of any class.
# single_unit: erx_usb_bot_core sends no Get Max LUN and serves unit 0 alone.
# no_inquiry: erx_usb_bot_core sends no INQUIRY and answers it itself, a removable
#   direct-access unit.
# ignore_residue: erx_usb_bot_core takes the residue from what the data phase moved,
#   not from the CSW.
# ignore_csw_tag: erx_usb_bot_core takes a CSW whatever its tag.
# wrong_csw_signature: erx_usb_bot_core takes a CSW signed usb_bot_csw_signature_dbx1
#   or usb_bot_csw_signature_olympus as signed usb_bot_csw_signature.
# no_sync_cache: erx_usb_bot_core sends no SYNCHRONIZE CACHE; the medium cannot flush.
# capacity_is_count: erx_usb_bot_core reads READ CAPACITY(10)'s last block, below all
#   ones, as the count of blocks, one more than the last.
enum usb_quirk u8
  value force_bot 0
  value force_scsi 1
  value single_unit 2
  value no_inquiry 3
  value ignore_residue 4
  value ignore_csw_tag 5
  value wrong_csw_signature 6
  value no_sync_cache 7
  value capacity_is_count 8
enum usb_bot_status u8
  value good 0
  value failed 1
  value phase_error 2
# The recovery ladder's bounds (design-usb.md, "Recovery"), erXos's choice: one attempt of
# a command waits this long on its bulk transfers before its transfer is cancelled; each
# step of recovery, a pipe reset or a class request, is failed this long after it was
# sent, as FreeBSD bounds each of its reset transfers (umass.c:519, 5000 ms); and an
# operation is sent again at most this many times, as FreeBSD's disk driver retries
# (scsi_da.c:1573, DA_DEFAULT_RETRY 4), within blk_in_flight_ms (blk.txt).
limit usb_bot_attempt_ms u32 8000
limit usb_bot_recovery_step_ms u32 5000
limit usb_bot_retries u32 4
# How often a unit is polled with TEST UNIT READY until its medium is ready, and a
# removable unit's medium after (design-usb.md, "The SCSI subset"), as FreeBSD's disk
# driver polls (scsi_da.c:3060-3065, as the design cites it).
limit usb_bot_poll_ms u32 3000

# The data each command that reads data asks for: INQUIRY's standard 36 bytes
# (SHORT_INQUIRY_LENGTH, scsi_all.h:2199), fixed-format sense's 18 (SSD_MIN_SIZE,
# scsi_all.h:3282), and READ CAPACITY(10)'s 8 and (16)'s 32 bytes (struct
# scsi_read_capacity_data and scsi_read_capacity_data_long, scsi_all.h:2958-2992).
limit scsi_inquiry_length u32 36
limit scsi_sense_length u32 18
limit scsi_read_capacity_10_length u32 8
limit scsi_read_capacity_16_length u32 32

# The fewest relevant bytes INQUIRY's decoder takes (design-usb.md, "The BOT engine"),
# its standard header, for the device type and the removable bit. The other decoders'
# minimums are the fields they read, which their binary patterns spell: INQUIRY's
# vendor[8] at byte 8 and product[16] at byte 16, taken when its first 32 bytes are
# there (scsi_all.h:2294-2297); fixed-format sense through the additional sense code and
# its qualifier in bytes 12 and 13, 14 bytes (scsi_all.h:3322-3326); and READ
# CAPACITY(10)'s 4-byte and (16)'s 8-byte last block address, each followed by the
# 4-byte block length, 8 and 12 bytes (scsi_all.h:2959-2968).
limit scsi_inquiry_min u32 5
# The one block length the transport serves (design-usb.md, "The SCSI subset"), erXos's
# choice; a medium of any other is not served.
limit scsi_block_length u32 512
