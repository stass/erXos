# The device-secret page ("Device secrets" in docs/design-platforms.md):
# written once in the boot-only section, before any reader exists, and read
# only through the page's own mapping. It is a shared layout, not a wire one,
# so the generator emits no decoder that would copy the secret into a frame its
# reader could not zero. status holds a device_secret_status; any other status
# than present means every byte of secret is zero, and the rest of the page is
# zero. Each label is public: the kernel derives the device secret with

# Capability values and tables (docs/design-kernel.md, "Objects and
# capabilities"). A value holds its slot's index in bits 0-23 and the
# slot's generation in bits 24-63; slot 0 is never issued, so value 0 names
# nothing, and cap_null spells it. A table grows a page of cap_page_slots
# slots at a time up to its space's limit, at most caps_max slots, slot 0
# included. cap_self and cap_self_thread are reserved values whose slot bits
# lie above caps_max, so no slot can ever bear them, and the one lookup
# resolves them to the calling thread's space and thread.
limit cap_null u64 0
limit cap_self u64 0xffffff
limit cap_self_thread u64 0xfffffe
limit cap_slot_bits u32 24
limit cap_generation_bits u32 40
limit caps_max u32 1048576
limit cap_page_slots u32 64

# A capability slot, a sequence-counted record that lookups read without a
# lock while its table's writer changes it ("Reading a slot without a
# lock"). object is the object store's nonzero handle while the slot is
# live, 0 otherwise; type is the object's type, fixed for its life; next
# links the table's free list by slot index. table is the address of the
# table the slot belongs to, fixed when its page is added. obj_prev and
# obj_next are the addresses of the slots before and after it on its
# object's slot list, 0 at either end, which every operation that
# installs or clears a slot changes under the object's capability lock;
# nothing reads them without it, and a lookup never reads these three. seq never
# resets: its terminal value retires the slot for good, and a lookup
# reads it as a stale capability. It is private: only kernel/core/cap.c and its host
# tests see its members (K1), through build/gen/kernel/generated/cap_slot_impl.h.
enum cap_slot_state u32
  value free 0
  value reserved 1
  value live 2
  value retired 3
