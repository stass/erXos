# Capability values and tables (docs/design-kernel.md, "Objects and
# capabilities"). A value holds its slot's index in bits 0-23 and the
# slot's generation in bits 24-63; slot 0 is never issued, so value 0 names
# nothing. A table grows a page of cap_page_slots slots at a time up to its
# space's limit, at most caps_max slots, slot 0 included.
limit cap_slot_bits u32 24
limit cap_generation_bits u32 40
limit caps_max u32 1048576
limit cap_page_slots u32 64

# Object types: the rights each type's capability can carry, and its pass
# set, the rights it keeps when it passes to a peer or a fork copies it;
# an empty pass set means it never passes between peers ("Rights").
type budget 1
  rights alloc,spaces,code,carve,inspect,dup,give,destroy
  pass none
type space 2
  rights map,threads,caps,stop,inspect,dup,give,destroy
  pass none
type thread 3
  rights control,configure,inspect,dup,give,destroy
  pass none
type group 4
  rights configure,attach,signal,inspect,dup,give,destroy
  pass none
type memory 5
  rights read,write,exec,map,jit,phys,supply,child,carve,system,inspect,dup,give,destroy
  pass read,write,map,child,inspect,dup,give
type gate 6
  rights call,serve,configure,inspect,dup,give,destroy
  pass call,serve,inspect,dup,give

  rights trigger,bind,ack,configure,inspect,dup,give,destroy
  pass trigger,inspect,dup,give
type resource 8
  rights carve,power,trace,entropy,inspect,dup,give
  pass none
type dma 9
  rights map,bind,inspect,dup,give,destroy
  pass none

# A capability slot, a sequence-counted record that lookups read without a
# lock while its table's writer changes it ("Reading a slot without a
# lock"). object is the object store's nonzero handle while the slot is
# live, 0 otherwise; type is the object's type, fixed for its life; next
# links the table's free list by slot index. seq never resets: its
# terminal value retires the slot for good, and a lookup reads it as a
# stale capability. It is private: only kernel/core/cap.c and its host
# tests see its members (K1), through kernel/generated/cap_slot_impl.h.
enum cap_slot_state u32
  value free 0
  value reserved 1
  value live 2
  value retired 3
layout erx_cap_slot 1 shared 8 64
  private kernel/core/cap.c
  field object u64 0
  field generation u64 8
  field seq u64 16
  field rights u32 24
  field type u32 28
  field state u32 32
  field next u32 36
  padding 40 24
  atomic object load acquire store release
  atomic generation load acquire store release
  atomic seq load acquire store release
  atomic rights load acquire store release
  atomic type load acquire store release
  atomic state load acquire store release
  atomic next load acquire store release
  writer object cap_table
  writer generation cap_table
  writer seq cap_table
  writer rights cap_table
  writer type cap_table
  writer state cap_table
  writer next cap_table
  owner object kernel all
  owner generation kernel all
  owner seq kernel all
  owner rights kernel all
  owner type kernel all
  owner state kernel all
  owner next kernel all
  visibility object internal
  visibility generation internal
  visibility seq internal
  visibility rights internal
  visibility type internal
  visibility state internal
  visibility next internal
  sequence seq terminal 0xffffffffffffffff exhaustion @stale recovery none
