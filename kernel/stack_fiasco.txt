# The C17 cores' share of Fiasco's thread block on the stopgap, read by the
# stack proof (tools/lib/erx/stack_proof_input.rb, docs/design-stopgap.md,
# "The stack proof"). The proof bounds the cores' depth from each of their
# exported functions against the allowance of its role; Fiasco's and the
# glue's use of the rest of the block is measured by the conformance
# suite's watermark, never proven.
#
#   block BYTES             the thread block (THREAD_BLOCK_SIZE, 8 KiB under
#                           CONFIG_CONTEXT_8K)
#   allowance ROLE BYTES    the cores' share for entries of ROLE: call, for
#                           every exported function called on a thread's
#                           kernel path; tick, for those the tick hook calls
#   tick NAME               an entry the tick hook calls
#   leaf NAME BYTES         a function outside the cores that a core calls,
#                           counted as a leaf of this declared cost

block 8192

allowance call 2048
allowance tick 1024

tick erx_watchdog_step

# Fiasco's own memcpy and memset (fiasco/kernel/src/lib/libc/src/string/),
# which the cores' archive may leave undefined. Both are leaves with no
# frame under GCC 16.2.0 for aarch64-elf at -O2 -ffreestanding
# -fstack-usage (measured 2026-11-03 on the macOS development host); the
# declared cost leaves room for Fiasco's own flags, and the watermark run
# catches a cost declared too low.
leaf memcpy 64
leaf memset 64

# The budget core's lock effects (include/erx/budget.h), which the glue
# provides over the core's ticket lock with interrupt masking. Not yet
# measured: the declared cost is the budget the glue's two functions must
# meet, and the watermark run catches a cost declared too low.
leaf erx_effect_budget_lock 128
leaf erx_effect_budget_unlock 128

# The debug log's effects (include/erx/core/effects.h), which the glue
# provides over a Spin_lock, Timer::system_clock and Fiasco's console. The
# costs declared are unmeasured budgets those functions must meet, the
# console's being the largest, through its mux and UART driver; the watermark
# run catches a cost declared too low.
leaf erx_effect_log_lock 128
leaf erx_effect_log_unlock 128
leaf erx_effect_clock_ns 128
leaf erx_effect_console_write 512
