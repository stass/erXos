# The system domain serves inspect and a session holds it, so both take inspect's
# floors (abi/services/inspect.txt, property 16): a max_frame of
# inspect_max_frame_min, 512 KiB, a max_message_heap of at least
# inspect_message_heap_min, 131072 words, and a funnel_depth of at least
# inspect_funnel_depth_min. A record buffer grows only as a frame's bytes arrive, but

# data: one erx_fs grant of /data's tree for each grantee, read-write and bound to the
# grantee's domain id. Each grantee mounts it at /data (docs/design-files.md, "A facet's
# mount"); the system domain's travels on the storage domain's system connection. Its max_request is files_max_frame_min (property 16), not the
# largest 9P msize, 65560, since a request is measured as its encoded Erlang term, which
# for a 64 KiB write with its envelope is longer than that; in_flight caps the requests the grant holds at once; max_heap_size, in
# words (2 MiB), bounds the grant's process, which holds up to in_flight write requests of
# 64 KiB with their data. reductions and period_ns bind no erx_fs grant. The storage
# domain takes two connections, root's session and one more while a session is replaced,
# and root's session four: data, net, and one more for each while its server is
# replaced; a domain's system connection counts against no quota (life_connections,
# liberx/system/dom_life.c). The network domain, like the storage domain, takes two,
# and the display domain, whose display no facet grants, holds its system connection
# alone, with a quota of none. Each board's machine record reckons its max_connections
# from these rows.
service data
  server @domain.storage
  interface files
facet data
  grantee system
  service @service.data
  operations none
  max_request 131072
  in_flight 16
  reductions 100000
  period_ns 1000000000
  max_heap_size 262144

# display: the display domain's window tree (docs/design-display.md, "A display's
# session"), of the windows interface, and its display record. No facet grants it and no
# start row names it, so the display shows its empty layout's background. The policy is
# single and the idle times 0, which the display_domain suite boots
# (DisplayDomain.manifest); the domain library passes the idle times to the domain's
# environment, and nothing in the display domain reads them yet, so 0 has no meaning
# of its own here.
service display
  server @domain.display
  interface windows
display display
  policy single
  idle_dim_s 0
  idle_off_s 0
# log: the system log's write interface (docs/design-log.md, "The log facet"), served by
# the system domain, where erx_syslog_sup registers as its server (erx_facet:serve/2).
# No facet row grants it, so no other domain holds a log facet. A row that grants it
# must give a max_request that holds one full write of log_write_max records of
# log_write_text_max bytes, or the facet drops a legal write unanswered (TODO.md,
# "log facet").
service log
  server system
  interface log
# inspect: the inspector's queries for each session made from root's template
# (docs/design-live.md, "The inspect service"), served by the system domain, so it
# travels on the session's system connection and takes none of max_connections. A
# request is a path and its options within 4 KiB; in_flight is the design's default of
# two query workers at once; max_heap_size, in words (512 KiB), bounds the facet's
# process, which holds the requests in flight, not the answers.
service inspect
  server system
  interface inspect

  grantee @session.system
  service @service.inspect
  operations none
  max_request 4096
  in_flight 2
  reductions 100000
  period_ns 1000000000
  max_heap_size 65536
# apps: the start service's tree, /apps/ctl and /apps/rc.conf (docs/design-apps.md,
# "Interfaces"), for each session made from root's template: one erx_fs grant served by
# the system domain, bound to the session and ending with it, so it travels on the
# session's system connection and takes none of max_connections. Its max_request is
# files_max_frame_min (property 16), as for every files grant; a ctl write is one line of at most
# 4 KiB and rc.conf at most 64 KiB, so in_flight is small; max_heap_size, in words
# (1 MiB), bounds the grant's process, which holds rc.conf's text while it answers.
