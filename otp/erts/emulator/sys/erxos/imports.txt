# What ERTS's objects may import from outside themselves and liberx. Read by
# make check-runtime-seam; the rule is in docs/design-runtime.md, "What ERTS
# imports". Names are separated by white space; a section is a group that can
# be dropped whole.
#
# The sections up to [libsystem] are the list the design keeps. Every [stock_*]
# section is the stock Unix layer sys/erxos was copied from. Its comment says what
# empties it, and that change deletes the section. The check reports a listed name
# that nothing imports.

# The crash dump (edit 34), erlang:display and halt's output in bif.c, the terminal's
# stderr and the wake pipes of sys.c, erl_printf's stdio, and ethread's waits. Each
# goes with the first native boot, or for prim_tty's with the console.
[stock_descriptors]
close fclose fdopen fileno open read write writev

# What sys.c keeps of a Unix process: the signal dispatcher and its pipes on Darwin's
# main thread, nice, alarm, the node id's host name and pid, and uname; ddll's dlopen;
# the select sleeps of beam_debug, erl_bif_info and erl_misc_utils; and ethread's
# signal masks. Each is answered at the first native boot (TODO.md).
[stock_process]
__darwin_check_fd_set_overflow
_exit alarm dlclose dlerror dlopen dlsym fcntl gethostname getpid ioctl nice pipe
pthread_kill pthread_sigmask select sigaction sigwait uname

# erx_stdio_nif.c alone, hosted mode's console device: the only object of the runtime
# that may use these for a host terminal.
[only_erx_stdio_nif]

# Stock callers not yet traced: each caller goes with its layer, or the name joins
# the design's list with a reason.
[stock_unsorted]
